# PDSCH + TDL 다중경로 채널 + HARQ IR 시뮬레이션
# 3GPP TS 38.901 TDL 채널 모델 + TS 38.212 HARQ IR

PHYSICAL_CHANNEL = TDL_HARQ

# ── MCS 설정 ──────────────────────────────────────────────────────────────────
MCS_INDEX  = 10
MCS_TABLE  = TABLE1           # TABLE1 (64QAM max) / TABLE2 (256QAM max)
TB_SIZE    = 0                # 0 = auto (num_data_RE * code_rate 기반)

# ── Resource Grid ─────────────────────────────────────────────────────────────
BANDWIDTH_MHZ    = 20
SCS_KHZ          = 30         # 서브캐리어 간격 [kHz]
NUM_RB           = 25         # Resource Block 수
NFFT             = 1024
CP_LENGTH_NORMAL = 72         # Normal CP (144 * NFFT/2048 samples at 30kHz)

# ── TDL 채널 모델 (TS 38.901) ─────────────────────────────────────────────────
# TDL_A : NLOS, 23-tap, 단거리 산란 (Urban micro, indoor)
# TDL_C : NLOS, 24-tap, 장거리 산란 (Urban macro NLOS)
# TDL_D : LOS,  13-tap, K=13.3 dB  (Suburban / rural LOS)
TDL_MODEL    = TDL_A
TDL_DS_RMS_NS = 30.0          # RMS delay spread [ns]
                               # TDL-A 권장: 10~300 ns
                               # TDL-C 권장: 100~1000 ns
                               # TDL-D 권장: 10~300 ns

# ── 도플러 (Jakes AR(1) 모델) ─────────────────────────────────────────────────
# fd [Hz] = v[m/s] * fc[Hz] / c[m/s]
# v=3km/h,  fc=3.5GHz → fd ≈ 9.7 Hz    (보행자)
# v=30km/h, fc=3.5GHz → fd ≈ 97 Hz     (저속 차량)
# v=120km/h,fc=3.5GHz → fd ≈ 389 Hz    (고속 차량)
# HARQ round간 IID 채널 가정 (tdl_new_realization 사용)
DOPPLER_HZ   = 0.0            # 0 = static channel

# ── HARQ 설정 ─────────────────────────────────────────────────────────────────
HARQ_MAX_ROUNDS = 4           # 최대 재전송 횟수 (RV=0,1,2,3)

# ── MIMO 설정 ─────────────────────────────────────────────────────────────────
# MRC (Maximal Ratio Combining): 각 Rx 안테나는 독립 TDL 채널
# 1Rx : SISO (기준)
# 2Rx : 2-branch MRC, 이론 diversity gain ≈ 3 dB at BLER=1%
# 4Rx : 4-branch MRC
NUM_RX_ANT = 2

# ── 채널 추정 ─────────────────────────────────────────────────────────────────
# LS    : DMRS LS 추정 + 선형 보간 (기존 방식)
# LMMSE : W = R_dp·(R_pp + N0·I)^{-1}, 주파수 상관 exp(-2π·τ_rms·Δf·|Δk|)
#         → 저SNR / 큰 delay spread에서 LS 대비 1~3 dB 이득
CHANNEL_EST = LMMSE

# ── 등화기 ────────────────────────────────────────────────────────────────────
# MMSE : per-SC 노이즈 분산 LLR (기본, 권장) → 주파수 선택적 채널에서 최적
# ZF   : 평균 채널 파워 기반 flat LLR → 비교 실험용 (성능 열화 확인)
EQUALIZER = MMSE

# ── SNR 스윕 ──────────────────────────────────────────────────────────────────
SNR_START = -4
SNR_END   = 14
SNR_STEP  = 2

# ── 시뮬레이션 반복 수 ────────────────────────────────────────────────────────
# 정확한 BLER 곡선(~1e-2): NUM_TRIALS = 1000~5000
# 논문급 정확도(~1e-3):    NUM_TRIALS = 10000 이상
NUM_TRIALS = 2000

# ── Monte Carlo 조기 종료 ─────────────────────────────────────────────────────
# 해당 SNR 포인트에서 1라운드 블록 에러가 이 수에 도달하면 시뮬 중단
# 200 → BLER 신뢰구간 약 ±2% (95% CI)
# 500 → BLER 신뢰구간 약 ±1%
MIN_BLOCK_ERRORS = 200

# ── CSV 출력 ──────────────────────────────────────────────────────────────────
OUTPUT_CSV  = 1
OUTPUT_FILE = results/result_tdl_harq.csv
