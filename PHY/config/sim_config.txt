# 5G PHY Link Level Simulator Configuration
# L1 Parameters

# Bandwidth (MHz): 5, 10, 15, 20, 25, 30, 40, 50, 60, 80, 100
BANDWIDTH_MHZ = 100

# Subcarrier Spacing (kHz): 15, 30, 60, 120
SCS_KHZ = 30

# Number of Resource Blocks (0 = auto from BW/SCS)
NUM_RB = 273

# FFT Size (0 = auto, or 128, 256, 512, 1024, 2048, 4096)
NFFT = 4096

# Cyclic Prefix Length (0 = auto per 3GPP TS 38.211)
# First symbol (0, 7): 160 * NFFT / 2048
# Normal symbols (1-6, 8-13): 144 * NFFT / 2048
CP_LENGTH_FIRST = 0
CP_LENGTH_NORMAL = 0

# Modulation: QPSK, 16QAM, 64QAM, 256QAM
MODULATION = QPSK

# Channel Coding: NONE, LDPC, POLAR
CODING = LDPC

# Code Rate (for LDPC): 0.5, 0.67, 0.75, 0.83
CODE_RATE = 0.5

# Number of OFDM Symbols per simulation
NUM_OFDM_SYMBOLS = 50

# SNR Range (dB): START, END, STEP
SNR_START = 0
SNR_END = 25
SNR_STEP = 3

# Channel Model: AWGN, FLAT_FADING
CHANNEL_MODEL = AWGN

# Physical Channel: NONE, PBCH, PDCCH, PDSCH
PHYSICAL_CHANNEL = PDSCH

# MCS Index (0-28 for TABLE1, 0-27 for TABLE2)
MCS_INDEX = 10

# MCS Table: TABLE1 (TS 38.214 Table 5.1.3.1-1), TABLE2 (Table 5.1.3.1-2)
MCS_TABLE = TABLE1

# Number of simulation trials per SNR point
NUM_TRIALS = 1000

# DMRS + Channel Estimation (0=off, 1=on)
# When enabled: DMRS Type 1 pilots inserted at even subcarriers per RB,
# flat Rayleigh fading channel estimated via LS + linear interpolation,
# equalized with ZF before soft demodulation.
USE_DMRS = 1

# Equalizer type: ZF (Zero-Forcing) or MMSE (Minimum Mean Square Error)
# MMSE avoids noise amplification in deep fades by using biased output
# with per-subcarrier effective noise variance in LLR computation.
EQUALIZER = MMSE
